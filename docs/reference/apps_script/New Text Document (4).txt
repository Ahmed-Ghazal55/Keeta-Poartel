<script>
  const gasStatuses = ["نشط", "مراجعة المركبات", "موقوف", "لا يوجد"];

  function toggleIban(){
    const opt = document.getElementById("ibanOptions").value;
    document.getElementById("ibanInput").style.display = opt === "manual" ? "block" : "none";
  }

  function setLoading(btnId, state, text){
    const btn = document.getElementById(btnId);
    btn.disabled = state;
    btn.innerText = state ? text : btn.dataset.original;
  }

  document.getElementById("searchBtn").dataset.original = document.getElementById("searchBtn").innerText;
  document.getElementById("saveBtn").dataset.original = document.getElementById("saveBtn").innerText;

  function search(){
    const iqama = document.getElementById("iqama").value.trim();
    if(!iqama) return alert("برجاء إدخال رقم الإقامة");
    setLoading("searchBtn", true, "جاري البحث...");
    google.script.run
      .withSuccessHandler(fillForm)
      .withFailureHandler(err => {
        setLoading("searchBtn", false);
        alert("حدث خطأ: " + err.message);
      })
      .searchRider(iqama);
  }

  function fillForm(rider){
    setLoading("searchBtn", false);
    document.getElementById("mainForm").classList.remove("hidden");
    document.getElementById("saveBtn").style.display = "block";

    clearForm(false);

    if(rider && rider.found){
      document.getElementById("row").value = rider.row || "";
      document.getElementById("statusBox").innerText = "✅ تم العثور على المندوب ويمكن تعديل بياناته";
      document.getElementById("subTypeText").innerText = rider.subType || "-";
      document.getElementById("currentUserText").innerText = rider.currentUser || "غير موجود حاليا";
      document.getElementById("name").value = rider.name || "";
      document.getElementById("phone").value = rider.phone || "";
      document.getElementById("appPhone").value = rider.appPhone || "";
      setSelectValue("vehicleType", rider.vehicleType || "سيارة شركة");
      setSelectValue("nationality", rider.nationality || "");

      const gas = rider.gasCard || "";
      const matched = gasStatuses.find(s => gas.includes(s));
      document.getElementById("gasStatus").value = matched || "";
      document.getElementById("gasNumber").value = matched ? gas.replace(matched, "").replace(/[-–]/g, "").trim() : gas;

      const toolsArr = splitTools(rider.tools || "");
      document.querySelectorAll("#toolsGroup input").forEach(cb => cb.checked = toolsArr.includes(cb.value));

      const iban = rider.iban || "";
      if(iban === "CASH" || iban === "جمع الايبان لاحقاً"){
        document.getElementById("ibanOptions").value = iban;
        document.getElementById("ibanInput").value = "";
      } else {
        document.getElementById("ibanOptions").value = "manual";
        document.getElementById("ibanInput").value = iban;
      }
      toggleIban();
      document.getElementById("saveBtn").innerText = "💾 تحديث بيانات المندوب";
      document.getElementById("saveBtn").dataset.original = "💾 تحديث بيانات المندوب";
    } else {
      document.getElementById("statusBox").innerText = "➕ المندوب غير موجود، سيتم إنشاء سجل جديد";
      document.getElementById("subTypeText").innerText = "سيظهر تلقائيًا بعد الحفظ";
      document.getElementById("currentUserText").innerText = "غير موجود حاليا";
      document.getElementById("saveBtn").innerText = "➕ إضافة مندوب جديد";
      document.getElementById("saveBtn").dataset.original = "➕ إضافة مندوب جديد";
    }
  }

  function clearForm(keepIqama){
    if(!keepIqama) {}
    ["row","name","phone","appPhone","gasNumber","ibanInput"].forEach(id => document.getElementById(id).value = "");
    document.getElementById("vehicleType").value = "سيارة شركة";
    document.getElementById("nationality").value = "";
    document.getElementById("gasStatus").value = "";
    document.getElementById("ibanOptions").value = "manual";
    document.querySelectorAll("#toolsGroup input").forEach(cb => cb.checked = false);
    toggleIban();
  }

  function splitTools(text){
    return text.split(/,\s*/).map(x => x.trim()).filter(Boolean);
  }

  function setSelectValue(id, value){
    const el = document.getElementById(id);
    const exists = Array.from(el.options).some(o => o.value === value);
    if(value && !exists) el.add(new Option(value, value));
    el.value = value;
  }

  function save(){
    const selectedTools = Array.from(document.querySelectorAll("#toolsGroup input:checked")).map(cb => cb.value).join(", ");
    const ibanOpt = document.getElementById("ibanOptions").value;
    const iban = ibanOpt === "manual" ? document.getElementById("ibanInput").value.trim() : ibanOpt;
    const gasStatus = document.getElementById("gasStatus").value.trim();
    const gasNumber = document.getElementById("gasNumber").value.trim();
    const gasCard = gasStatus ? `${gasStatus} - ${gasNumber}`.trim() : gasNumber;

    const data = {
      iqama: document.getElementById("iqama").value.trim(),
      name: document.getElementById("name").value.trim(),
      phone: document.getElementById("phone").value.trim(),
      nationality: document.getElementById("nationality").value,
      appPhone: document.getElementById("appPhone").value.trim(),
      vehicleType: document.getElementById("vehicleType").value,
      gasCard: gasCard,
      tools: selectedTools,
      iban: iban
    };

    if(!data.iqama) return alert("رقم الإقامة مطلوب");
    if(!data.name) return alert("اسم المندوب مطلوب");

    setLoading("saveBtn", true, "جاري الحفظ...");
    google.script.run
      .withSuccessHandler(msg => {
        alert(msg);
        google.script.host.close();
      })
      .withFailureHandler(err => {
        setLoading("saveBtn", false);
        alert("حدث خطأ: " + err.message);
      })
      .saveRiderToMaster(data);
  }
</script>
