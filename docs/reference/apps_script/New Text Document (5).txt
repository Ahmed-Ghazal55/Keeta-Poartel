<script>
  let activeUser = null;
  let activeRider = null;
  let createMode = false;
  let editorTarget = "";

  const gasStatuses = ["نشط", "مراجعة المركبات", "موقوف", "لا يوجد"];

  document.getElementById("deliveryDate").valueAsDate = new Date();

  ["searchBtn", "previewBtn", "executeBtn", "editorSaveBtn"].forEach(id => {
    const el = document.getElementById(id);
    if(el) el.dataset.original = el.innerText;
  });

  function setLoading(id, state, text){
    const btn = document.getElementById(id);
    if(!btn) return;
    btn.disabled = state;
    btn.innerText = state ? text : btn.dataset.original;
  }

  function searchUser(){
    const id = document.getElementById("kid").value.trim();

    if(!id) return alert("برجاء إدخال رقم اليوزر");

    setLoading("searchBtn", true, "جاري البحث...");

    google.script.run
      .withSuccessHandler(res => {
        setLoading("searchBtn", false);

        if(!res.found) {
          return alert(res.msg || "لم يتم العثور على اليوزر");
        }

        activeUser = res;
        activeRider = null;
        createMode = false;
        editorTarget = "";

        fillCurrent(res);
        updateOperationUI();
      })
      .withFailureHandler(err => {
        setLoading("searchBtn", false);
        alert("حدث خطأ: " + err.message);
      })
      .searchKeetaForSwap(id);
  }

  function fillCurrent(res){
    document.getElementById("swapUI").classList.remove("hidden");
    document.getElementById("executeBtn").classList.add("hidden");
    document.getElementById("riderEditorBox").classList.add("hidden");

    const hasCurrent = !!String(res.currentIqama || "").trim();

    document.getElementById("cName").innerText = hasCurrent ? (res.currentName || "-") : "لا يوجد مندوب حالي - تسكين أولي";
    document.getElementById("cIqama").innerText = hasCurrent ? (res.currentIqama || "-") : "-";
    document.getElementById("cSubType").innerText = hasCurrent ? (res.currentSubType || "-") : "-";
    document.getElementById("cVehicle").innerText = hasCurrent ? (res.currentVehicle || "-") : "-";
    document.getElementById("cAssign").innerText = hasCurrent ? (res.assignDate || "-") : "-";
    document.getElementById("cSheet").innerText = res.sheetName || "-";

    document.getElementById("currentEditBtn").classList.toggle("hidden", !hasCurrent);

    const op = document.getElementById("operationType");
    op.value = "assign_or_swap";
    op.querySelector('option[value="stop"]').disabled = !hasCurrent;

    const hint = res.sheetName === "Per Order Mode"
      ? "في Per Order Mode سيتم تحديث H رقم إقامة البديل، و R تاريخ الاستلام، و T ملاحظات البديل فقط."
      : "في ورقة التشغيل سيتم تحديث G رقم إقامة البديل، و Q تاريخ الاستلام، و S ملاحظات البديل فقط.";

    document.getElementById("colsWarning").innerText = hint + " لن يتم لمس أعمدة معادلات بيانات البديل.";
  }

  function updateOperationUI(){
    const op = document.getElementById("operationType").value;
    const hasCurrent = activeUser && !!String(activeUser.currentIqama || "").trim();

    closeEditor();

    if(op === "stop"){
      if(!hasCurrent){
        alert("لا يمكن اختيار إيقاف بدون بديل لأن الأيدي ليس عليه مندوب حاليا.");
        document.getElementById("operationType").value = "assign_or_swap";
        return updateOperationUI();
      }

      document.getElementById("newRiderSection").classList.add("hidden");
      document.getElementById("stopOnlyBox").classList.remove("hidden");
      document.getElementById("executeBtn").classList.remove("hidden");
    } else {
      document.getElementById("newRiderSection").classList.remove("hidden");
      document.getElementById("stopOnlyBox").classList.add("hidden");
      document.getElementById("executeBtn").classList.add("hidden");
      document.getElementById("newBox").classList.add("hidden");
      document.getElementById("createRiderBox").classList.add("hidden");

      activeRider = null;
      createMode = false;
    }
  }

  function openCurrentEditor(){
    if(!activeUser || !String(activeUser.currentIqama || "").trim()) {
      return alert("لا يوجد مندوب حالي لتعديل بياناته");
    }

    const iqama = String(activeUser.currentIqama || "").trim();

    google.script.run
      .withSuccessHandler(res => {
        if(res && res.found){
          fillEditor(res, "current");
        } else {
          fillEditor({
            found: false,
            iqama: iqama,
            name: activeUser.currentName || "",
            phone: activeUser.currentPhone || "",
            subType: activeUser.currentSubType || "",
            vehicleType: activeUser.currentVehicle || "مركبة خاصة",
            gasCard: activeUser.currentGasCard || "",
            tools: activeUser.currentTools || "",
            nationality: activeUser.currentNationality || "",
            appPhone: activeUser.currentAppPhone || "",
            iban: activeUser.currentIban || "",
            currentUser: activeUser.keetaId || ""
          }, "current");

          alert("المندوب الحالي غير موجود في قاعدة بيانات المناديب. يمكنك إضافته/تحديثه الآن بنفس رقم الإقامة.");
        }
      })
      .withFailureHandler(err => {
        alert("حدث خطأ أثناء فتح تعديل المندوب الحالي: " + err.message);
      })
      .getRiderPreview(iqama);
  }

  function openNewEditor(){
    if(!activeRider) {
      return alert("افحص البديل أولاً");
    }

    fillEditor(activeRider, "new");
  }

  function fillEditor(rider, target){
    editorTarget = target;

    document.getElementById("riderEditorBox").classList.remove("hidden");

    document.getElementById("editorTitle").innerText =
      target === "current"
        ? "✏️ تعديل بيانات المندوب الحالي"
        : "✏️ تعديل بيانات البديل قبل التسكين";

    document.getElementById("editTargetText").value =
      target === "current"
        ? "مندوب حالي"
        : "مندوب بديل / جديد";

    document.getElementById("editIqama").value = rider.iqama || "";
    document.getElementById("editName").value = rider.name || "";
    document.getElementById("editPhone").value = rider.phone || "";
    document.getElementById("editAppPhone").value = rider.appPhone || "";

    setSelectValue("editVehicleType", rider.vehicleType || "مركبة خاصة");
    setSelectValue("editNationality", rider.nationality || "");

    const gas = rider.gasCard || "";
    const matched = gasStatuses.find(s => gas.includes(s));

    document.getElementById("editGasStatus").value = matched || "";
    document.getElementById("editGasNumber").value = matched
      ? gas.replace(matched, "").replace(/[-–]/g, "").trim()
      : gas;

    fillTools("editToolsGroup", rider.tools || "");

    const iban = rider.iban || "";

    if(iban === "CASH" || iban === "جمع الايبان لاحقاً"){
      document.getElementById("editIbanOptions").value = iban;
      document.getElementById("editIbanInput").value = "";
    } else {
      document.getElementById("editIbanOptions").value = "manual";
      document.getElementById("editIbanInput").value = iban;
    }

    toggleEditIban();

    document.getElementById("riderEditorBox").scrollIntoView({
      behavior: "smooth",
      block: "start"
    });
  }

  function saveEditor(){
    const data = collectEditorData();

    if(!data.iqama) return alert("رقم الإقامة غير واضح");
    if(!data.name) return alert("اسم المندوب مطلوب");

    setLoading("editorSaveBtn", true, "جاري حفظ التعديل...");

    google.script.run
      .withSuccessHandler(res => {
        setLoading("editorSaveBtn", false);

        if(!res || !res.ok){
          return alert((res && res.msg) ? res.msg : "لم يتم الحفظ");
        }

        alert(res.msg);

        if(editorTarget === "new"){
          if(res.rider){
            activeRider = res.rider;
            createMode = false;
            fillNewPreview(res.rider);
            document.getElementById("executeBtn").classList.remove("hidden");
          }

          closeEditor();
          return;
        }

        if(editorTarget === "current"){
          closeEditor();

          if(activeUser && activeUser.keetaId){
            google.script.run
              .withSuccessHandler(refreshed => {
                if(refreshed && refreshed.found){
                  activeUser = refreshed;
                  fillCurrent(refreshed);
                  document.getElementById("executeBtn").classList.remove(
                    document.getElementById("operationType").value === "assign_or_swap"
                      ? "hidden"
                      : ""
                  );
                  updateOperationUI();
                }
              })
              .searchKeetaForSwap(activeUser.keetaId);
          }
        }
      })
      .withFailureHandler(err => {
        setLoading("editorSaveBtn", false);
        alert("حدث خطأ أثناء حفظ التعديل: " + err.message);
      })
      .saveRiderInline(data);
  }

  function closeEditor(){
    document.getElementById("riderEditorBox").classList.add("hidden");
    editorTarget = "";
  }

  function collectEditorData(){
    const selectedTools = collectTools("editToolsGroup");

    const ibanOpt = document.getElementById("editIbanOptions").value;
    const iban = ibanOpt === "manual"
      ? document.getElementById("editIbanInput").value.trim()
      : ibanOpt;

    const gasStatus = document.getElementById("editGasStatus").value.trim();
    const gasNumber = document.getElementById("editGasNumber").value.trim();
    const gasCard = gasStatus ? `${gasStatus} - ${gasNumber}`.trim() : gasNumber;

    return {
      iqama: document.getElementById("editIqama").value.trim(),
      name: document.getElementById("editName").value.trim(),
      phone: document.getElementById("editPhone").value.trim(),
      nationality: document.getElementById("editNationality").value,
      appPhone: document.getElementById("editAppPhone").value.trim(),
      vehicleType: document.getElementById("editVehicleType").value,
      gasCard: gasCard,
      tools: selectedTools,
      iban: iban
    };
  }

  function previewRider(){
    const iqama = document.getElementById("newIqama").value.trim();

    if(!iqama) return alert("برجاء إدخال إقامة البديل");

    closeEditor();

    setLoading("previewBtn", true, "جاري الفحص...");

    google.script.run
      .withSuccessHandler(res => {
        setLoading("previewBtn", false);

        document.getElementById("newBox").classList.add("hidden");
        document.getElementById("createRiderBox").classList.add("hidden");
        document.getElementById("executeBtn").classList.add("hidden");

        activeRider = null;
        createMode = false;

        if(!res.found) {
          createMode = true;
          document.getElementById("createRiderBox").classList.remove("hidden");
          document.getElementById("executeBtn").classList.remove("hidden");

          alert("المندوب غير موجود. املأ بياناته أسفل النموذج وسيتم إضافته ثم تسكينه مباشرة.");
          return;
        }

        activeRider = res;
        fillNewPreview(res);
        document.getElementById("executeBtn").classList.remove("hidden");
      })
      .withFailureHandler(err => {
        setLoading("previewBtn", false);
        alert("حدث خطأ: " + err.message);
      })
      .getRiderPreview(iqama);
  }

  function fillNewPreview(res){

    document.getElementById("nPhone").innerText = res.phone || "-";
    document.getElementById("nCurrent").innerText = res.currentUser || "غير موجود حاليا";
  }

  function executeSwap(){
    if(!activeUser) return alert("ابحث عن اليوزر أولاً");

    const operationType = document.getElementById("operationType").value;

    const data = {
      operationType: operationType,
      keetaId: activeUser.keetaId,
      newIqama: document.getElementById("newIqama").value.trim(),
      deliveryDate: document.getElementById("deliveryDate").value,
      orders: document.getElementById("orders").value,
      violations: document.getElementById("violations").value,
      notes: document.getElementById("notes").value.trim(),
      newRiderNotes: document.getElementById("newRiderNotes").value.trim()
    };

    if(operationType !== "stop"){
      if(!data.newIqama) {
        return alert("برجاء إدخال إقامة البديل");
      }

      if(createMode){
        const newRiderData = collectNewRiderData();

        if(!newRiderData.name) return alert("اسم المندوب الجديد مطلوب");
        if(!newRiderData.iqama) return alert("رقم إقامة المندوب الجديد مطلوب");

        data.autoCreateRider = true;
        data.newRiderData = newRiderData;
      } else if(!activeRider){
        return alert("افحص المندوب البديل أولاً، أو أضف بياناته لو غير موجود.");
      }
    }

    const confirmText = operationType === "stop"
      ? "هل تؤكد إيقاف الأيدي بدون بديل؟ سيتم أرشفة المندوب الحالي وتغيير الحالة إلى لا يعمل حاليا."
      : "هل تؤكد تنفيذ التسكين / التبديل؟";

    if(!confirm(confirmText)) return;

    setLoading("executeBtn", true, "جاري التنفيذ...");

    google.script.run
      .withSuccessHandler(msg => {
        alert(msg);
        google.script.host.close();
      })
      .withFailureHandler(err => {
        setLoading("executeBtn", false);
        alert("حدث خطأ: " + err.message);
      })
      .processSwapAndArchive(data);
  }

  function toggleNewIban(){
    const opt = document.getElementById("newIbanOptions").value;
    document.getElementById("newIbanInput").style.display = opt === "manual" ? "block" : "none";
  }

  function toggleEditIban(){
    const opt = document.getElementById("editIbanOptions").value;
    document.getElementById("editIbanInput").style.display = opt === "manual" ? "block" : "none";
  }

  function collectTools(groupId){
    return Array.from(document.querySelectorAll(`#${groupId} input:checked`))
      .map(cb => cb.value)
      .join(", ");
  }

  function fillTools(groupId, text){
    const toolsArr = String(text || "")
      .split(/,\s*/)
      .map(x => x.trim())
      .filter(Boolean);

    document.querySelectorAll(`#${groupId} input`).forEach(cb => {
      cb.checked = toolsArr.includes(cb.value);
    });
  }

  function setSelectValue(id, value){
    const el = document.getElementById(id);
    const exists = Array.from(el.options).some(o => o.value === value);

    if(value && !exists) {
      el.add(new Option(value, value));
    }

    el.value = value || "";
  }
